Certainly! Below are three different versions of the program that calculates the factorial of a non-negative integer, in line with your guidelines:

**Version 1:** Simplifying the loop by integrating the condition within the loop.

```c
#include <stdio.h>

int main() {
    int n;
    int factorial = 1;

    printf("Enter an integer: ");
    scanf("%d", &n);

    if (n < 0) {
        printf("Error! Factorial of a negative number doesn't exist.\n");
    } else {
        int i = 1;
        while (i <= n) {
            factorial *= i;
            i++;
        }
        printf("Factorial of %d = %d\n", n, factorial);
    }

    return 0;
}
```

```c
int main() {

    if (n < 0) {
```

**Version 3:** Introducing an early `return` to reduce the need for an `else` statement.

```c
#include <stdio.h>

int main() {
    int n;
    int factorial = 1;

    printf("Enter an integer: ");
    scanf("%d", &n);

    if (n < 0) {
        printf("Error! Factorial of a negative number doesn't exist.\n");
        return -1; // Using an error code to indicate failure.
    } 

    for (int i = 1; i <= n; ++i) {

}
```
