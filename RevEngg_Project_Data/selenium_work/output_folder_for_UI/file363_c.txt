### Version 1: Compact Version

This version combines reading and printing into the main function for a more compact approach.

```c
#include <stdio.h>

int main() {
    int arr[10];
    int i;

    // Reading array elements
    for(i = 0; i < 10; i++) {
        printf("Enter element %d: ", i);
        scanf("%d", &arr[i]);
    }

    // Printing array elements
    printf("Array elements are:\n");
    for(i = 0; i < 10; i++) {
        printf("arr[%d]: %d\n", i, arr[i]);
    }

    return 0;
}
```

### Version 2: While Loop Variant

This version utilizes `while` loops for reading and printing array elements instead of `for` loops.

```c
#include <stdio.h>

int main() {
    int arr[10];
    int i = 0;

    // Reading array elements using while loop
    while(i < 10) {
        printf("Enter element %d: ", i);
        scanf("%d", &arr[i]);
        i++;
    }

    i = 0; // Resetting the counter for printing
    printf("Array elements are:\n");
    while(i < 10) {
        printf("arr[%d]: %d\n", i, arr[i]);
        i++;
    }

    return 0;
}
```

### Version 3: Step Increment in Loop

This version introduces a step increment in the loop for a different iteration approach.

```c
#include <stdio.h>

int main() {
    int arr[10];
    int step = 2; // Step size for reading and printing

    // Reading array elements with step
    for(int i = 0; i < 10; i += step) {
        printf("Enter element %d: ", i);
        scanf("%d", &arr[i]);
        if(i + 1 < 10) {
            printf("Enter element %d: ", i+1);
            scanf("%d", &arr[i+1]);
        }
    }

    // Printing array elements
    printf("Array elements are:\n");
    for(int i = 0; i < 10; i++) {
        printf("arr[%d]: %d\n", i, arr[i]);
    }

    return 0;
}
```

Each version introduces a slight variation in handling array operations, offering different ways to achieve similar outcomes without changing the essence of the original task.
