#include <stdio.h>

int main() {

        printf("Reverse Number is : 0\n"); // Assuming reverse of 0 and negative numbers as a special case
    } else {
        do {
            dig = n % 10;
            revNumber = revNumber * 10 + dig;
            n /= 10;

}
