### Version 1: Selection Sort

This version of the program uses the selection sort algorithm to sort the array in ascending order.

```c
#include <stdio.h>

int main() {
    int arr[100];
    int size, i, j, min_index, temp;

    printf("Enter size of array: ");
    scanf("%d", &size);

    printf("\nEnter elements in array: \n");
    for(i = 0; i < size; i++) {
        printf("\nEnter element %d: ", i+1);
        scanf("%d", &arr[i]);
    }

    for(i = 0; i < size - 1; i++) {
        min_index = i;
        for(j = i + 1; j < size; j++) {
            if(arr[j] < arr[min_index]) {
                min_index = j;
            }
        }
        if(min_index != i) {
            temp = arr[i];
            arr[i] = arr[min_index];
            arr[min_index] = temp;
        }
    }

    printf("\nElements of array in sorted ascending order: ");
    for(i = 0; i < size; i++) {
        printf(" %d ", arr[i]);
    }

    return 0;
}
```

### Version 2: Insertion Sort

This version uses the insertion sort algorithm for sorting the array elements in ascending order.

```c
#include <stdio.h>

int main() {
    int arr[100];
    int size, i, j, key;

    printf("Enter size of array: ");
    scanf("%d", &size);

    printf("\nEnter elements in array: \n");
    for(i = 0; i < size; i++) {
        printf("\nEnter element %d: ", i+1);
        scanf("%d", &arr[i]);
    }

    for(i = 1; i < size; i++) {
        key = arr[i];
        j = i - 1;

        while(j >= 0 && arr[j] > key) {
            arr[j + 1] = arr[j];
            j = j - 1;
        }
        arr[j + 1] = key;
    }

    printf("\nElements of array in sorted ascending order: ");
    for(i = 0; i < size; i++) {
        printf(" %d ", arr[i]);
    }

    return 0;
}
```

### Version 3: Bubble Sort

This version employs the bubble sort algorithm to sort the array elements in ascending order.

```c
#include <stdio.h>

int main() {
    int arr[100];
    int size, i, j, temp;

    printf("Enter size of array: ");
    scanf("%d", &size);

    printf("\nEnter elements in array: \n");
    for(i = 0; i < size; i++) {
        printf("\nEnter element %d: ", i+1);
        scanf("%d", &arr[i]);
    }

    for(i = 0; i < size - 1; i++) {
        for(j = 0; j < size - i - 1; j++) {
            if(arr[j] > arr[j + 1]) {
                temp = arr[j];
                arr[j] = arr[j + 1];
                arr[j + 1] = temp;
            }
        }
    }

    printf("\nElements of array in sorted ascending order: ");
    for(i = 0; i < size; i++) {
        printf(" %d ", arr[i]);
    }

    return 0;
}
```
