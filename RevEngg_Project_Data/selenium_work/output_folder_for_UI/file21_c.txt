### Version 1: Use Separate Variables for Sum and Average Calculation

```c
#include <stdio.h>

int main() {
    int num1, num2, totalSum;
    float average;

    printf("Enter the first number: ");
    scanf("%d", &num1);
    printf("Enter the second number: ");
    scanf("%d", &num2);

    totalSum = num1 + num2;
    average = totalSum / 2.0;

    printf("Sum of %d and %d is = %d\n", num1, num2, totalSum);
    printf("Average of %d and %d is = %.2f\n", num1, num2, average);

    return 0;
}
```

### Version 2: Use Compound Assignment Operators

```c
#include <stdio.h>

int main() {
    int num1, num2;
    int sum;
    float average;

    printf("Enter the first number: ");
    scanf("%d", &num1);
    printf("Enter the second number: ");
    scanf("%d", &num2);

    sum = 0;
    sum += num1;
    sum += num2;

    average = sum / 2.0f;

    printf("Sum of %d and %d is = %d\n", num1, num2, sum);
    printf("Average of %d and %d is = %.2f\n", num1, num2, average);

    return 0;
}
```

### Version 3: Direct Calculation in Variable Assignment

```c
#include <stdio.h>

int main() {

    scanf("%d", &x);
    printf("Enter the second number: ");

    sum = x + y; // Directly calculate sum in the assignment
    average = sum / 2; // Directly calculate average in the assignment

    printf("Sum of %d and %d is = %.0f\n", x, y, sum); // Print sum as float for consistency
    printf("Average of %d and %d is = %.2f\n", x, y, average);

}
```
