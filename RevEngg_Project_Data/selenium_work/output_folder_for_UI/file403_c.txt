### Version 1: Simplified Conditions and Messages

```c
#include <stdio.h>

int main() {
    int num;
    printf("Enter a number: ");
    scanf("%d", &num);

    if ((num % 7 == 0) || (num % 3 == 0)) {
        printf("The number %d is divisible by either 7 or 3.\n", num);
    } else {
        printf("The number %d is not divisible by either 7 or 3.\n", num);
    }

    return 0;
}
```

### Version 2: Using Flags for Clarity

```c
#include <stdio.h>

int main() {
    int num, isDivisible = 0;
    printf("Enter a number: ");
    scanf("%d", &num);

    if (num % 7 == 0) {
        isDivisible = 1;
    }
    if (num % 3 == 0) {
        isDivisible = 1;
    }

    if (isDivisible) {
        printf("The number %d is divisible by 7 or 3.\n", num);
    } else {
        printf("The number %d is not divisible by 7 or 3.\n", num);
    }

    return 0;
}
```

### Version 3: Nested Conditions for Extended Logic

```c
#include <stdio.h>

int main() {
    int num;
    printf("Enter a number: ");
    scanf("%d", &num);

    if (num % 7 == 0) {
        printf("The number %d is divisible by 7.\n", num);
        if (num % 3 == 0) {
            printf("Additionally, it is also divisible by 3.\n");
        }
    } else if (num % 3 == 0) {
        printf("The number %d is divisible by 3.\n", num);
    } else {
        printf("The number %d is neither divisible by 7 nor by 3.\n", num);
    }

    return 0;
}
```

Each of these versions approaches the problem with slightly different logic or formatting for the output. The core functionality remains the same, checking divisibility by 3 or 7, but demonstrates varied programming styles and choices for handling conditions.
