### Version 1: Direct Conversion with Literal Constant

```c
#include <stdio.h>

int main() {
    float powerInHorsepower, powerInWatts;
    printf("Enter power in Horsepower: ");
    scanf("%f", &powerInHorsepower);
    
    powerInWatts = powerInHorsepower * 746.0f;
    
    printf("Horsepower is equal to ");
    printf("%.2f", powerInWatts);
    printf(" Watts.\n");
    
    return 0;
}
```

### Version 2: Use of Intermediate Variable for Conversion Factor

```c
#include <stdio.h>

int main() {
    float powerHP, powerW;
    float conversionFactor;
    
    printf("Enter power in Horsepower: ");
    scanf("%f", &powerHP);
    
    conversionFactor = 746.0f;
    powerW = powerHP * conversionFactor;
    
    printf("Horsepower is equal to ");
    printf("%.2f", powerW);
    printf(" Watts.\n");
    
    return 0;
}
```
