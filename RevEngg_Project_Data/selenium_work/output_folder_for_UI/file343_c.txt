### Version 1: Simplified Variable Names and Direct Calculation in printf
```c
#include <stdio.h>

int main() {
    float b, s, p;

    printf("Enter base: ");
    scanf("%f", &b);
    printf("Enter side: ");
    scanf("%f", &s);

    p = 2 * (b + s);

    printf("Perimeter: %.2f\n", p);

    return 0;
}
```

### Version 2: Using Intermediate Variables for Readability
```c
#include <stdio.h>

int main() {
    float baseLength, sideLength;
    float totalBaseLength, totalSideLength;
    float perimeter;

    printf("Base length: ");
    scanf("%f", &baseLength);
    printf("Side length: ");
    scanf("%f", &sideLength);

    totalBaseLength = baseLength * 2;
    totalSideLength = sideLength * 2;

    perimeter = totalBaseLength + totalSideLength;

    printf("Perimeter: %.2f\n", perimeter);

    return 0;
}
```
