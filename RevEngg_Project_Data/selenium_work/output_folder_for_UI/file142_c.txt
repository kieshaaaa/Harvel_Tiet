### Version 1: Utilizing a `while` loop for finding divisors and calculating the sum

```c
#include <stdio.h>

int main() {
    int n, i, j, sum;
    int count = 0;

    printf("Enter an upper limit: ");
    scanf("%d", &n);

    printf("Perfect numbers between 1 and %d are: ", n);

    i = 1;
    while (i <= n) {
        sum = 0;
        j = 1;
        while (j <= i / 2) {
            if (i % j == 0) {
                sum += j;
            }
            j++;
        }
        if (sum == i) {
            printf("%d ", i);
            count++;
        }
        i++;
    }

    printf("\nTotal number of perfect numbers between 1 and %d: %d\n", n, count);

    return 0;
}
```

### Version 2: Using a `do-while` loop for iterating through numbers and their divisors

```c
#include <stdio.h>

int main() {
    int n, i, j, sum;
    int count = 0;

    printf("Enter an upper limit: ");
    scanf("%d", &n);

    printf("Perfect numbers between 1 and %d are: ", n);

    i = 1;
    do {
        sum = 0;
        j = 1;
        do {
            if (i % j == 0) {
                sum += j;
            }
            j++;
        } while (j <= i / 2);

        if (sum == i) {
            printf("%d ", i);
            count++;
        }
        i++;
    } while (i <= n);

    printf("\nTotal number of perfect numbers between 1 and %d: %d\n", n, count);

    return 0;
}
```

### Version 3: Refactoring the condition checks to minimize computation for odd numbers (since all known perfect numbers are even)

```c
#include <stdio.h>

int main() {
    int n, i, j, sum;
    int count = 0;

    printf("Enter an upper limit: ");
    scanf("%d", &n);

    printf("Perfect numbers between 1 and %d are: ", n);

    for (i = 2; i <= n; i += 2) { // Start from 2, increment by 2, skipping all odd numbers
        sum = 1; // 1 is always a divisor
        for (j = 2; j <= i / 2; j++) {
            if (i % j == 0) {
                sum += j;
            }
        }
        if (sum == i) {
            printf("%d ", i);
            count++;
        }
    }

    printf("\nTotal number of perfect numbers between 1 and %d: %d\n", n, count);

    return 0;
}
```
