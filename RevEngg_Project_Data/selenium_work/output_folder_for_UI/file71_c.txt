### Version 1: Using a while loop

```c
#include <stdio.h>
#include <stdlib.h>

int main() {
    int start, end;
    printf("Enter the starting number: ");
    scanf("%d", &start);
    printf("Enter the ending number: ");
    scanf("%d", &end);

    printf("Even numbers between %d to %d\n", start, end);

    int counter = start;
    while (counter <= end) {
        if (counter % 2 == 0) {
            printf("%d ", counter);
        }
        counter++;
    }

    return 0;
}
```

### Version 2: Skipping Odd Numbers Directly

```c
#include <stdio.h>
#include <stdlib.h>

int main() {
    int start, end;
    printf("Enter the starting number: ");
    scanf("%d", &start);
    printf("Enter the ending number: ");
    scanf("%d", &end);

    if (start % 2 != 0) start++; // Ensure start is even

    printf("Even numbers between %d to %d\n", start, end);

    for (int counter = start; counter <= end; counter += 2) {
        printf("%d ", counter);
    }

    return 0;
}
```

### Version 3: Using a do-while loop

```c
#include <stdio.h>
#include <stdlib.h>

int main() {
    int start, end;
    printf("Enter the starting number: ");
    scanf("%d", &start);
    printf("Enter the ending number: ");
    scanf("%d", &end);

    printf("Even numbers between %d to %d\n", start, end);

    int counter = start;
    do {
        if (counter % 2 == 0) {
            printf("%d ", counter);
        }
        counter++;
    } while (counter <= end);

    return 0;
}
```
