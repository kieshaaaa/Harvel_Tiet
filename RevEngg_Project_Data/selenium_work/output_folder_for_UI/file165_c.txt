### Version 1: Simplified Approach

```c
#include <stdio.h>

int main() {
    float gradians, degrees;
    printf("Enter angle in gradians: ");
    scanf("%f", &gradians);
    degrees = gradians * 0.9; // Direct calculation to convert gradians to degrees
    printf("Equivalent in degrees: ");
    printf("%f\n", degrees);
    return 0;
}
```

### Version 2: Using Separate Variables for Conversion Factor

```c
#include <stdio.h>

int main() {
    float gradians, degrees, conversionFactor;
    conversionFactor = 180.0 / 200.0; // Store the conversion factor in a variable
    printf("Enter angle in gradians: ");
    scanf("%f", &gradians);
    degrees = gradians * conversionFactor; // Use the conversion factor variable
    printf("Equivalent in degrees: ");
    printf("%f\n", degrees);
    return 0;
}
```

### Version 3: Prompting User with More Information

```c
#include <stdio.h>

int main() {
    float gradians, degrees;
    printf("This program converts gradians to degrees.\n");
    printf("Please enter the angle in gradians: ");
    scanf("%f", &gradians);
    degrees = gradians * 0.9; // Direct conversion
    printf("Angle in degrees: ");
    printf("%f\n", degrees);
    return 0;
}
```

Each version offers a slightly different approach or user experience but adheres to the constraints provided.
