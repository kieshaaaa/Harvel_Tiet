### Version 1: Using `strcasecmp` for Case-Insensitive Comparison

```c
#include<stdio.h>
#include<string.h>

int main() {
    char username[20];
    char password[20];
    
    printf("Enter the Username: ");
    scanf("%s", username);
    
    printf("Enter the password: ");
    scanf("%s", password);
    
    if(strcasecmp(username, "admin") == 0 && strcmp(password, "karunya") == 0) {
        printf("Login Successful \n");
    } else {
        printf("Invalid username or password\n");
    }
    
    return 0;
}
```

### Version 2: Manual Case Conversion for Case-Insensitive Comparison

```c
#include<stdio.h>
#include<string.h>
#include<ctype.h>

int main() {
    char username[20];
    char password[20];
    char lowerUsername[20] = {0};
    
    printf("Enter the Username: ");
    scanf("%s", username);
    
    printf("Enter the password: ");
    scanf("%s", password);
    
    // Convert entered username to lowercase for case-insensitive comparison
    for(int i = 0; username[i]; i++){
        lowerUsername[i] = tolower(username[i]);
    }
    
    if(strcmp(lowerUsername, "admin") == 0 && strcmp(password, "karunya") == 0) {
        printf("Login Successful \n");
    } else {
        printf("Invalid username or password\n");
    }
    
    return 0;
}
```

### Version 3: Hardcoded Case-Insensitive Check

```c
#include<stdio.h>
#include<string.h>

int main() {
    char username[20];
    char password[20];
    int isValid = 1; // Flag to check validity
    
    printf("Enter the Username: ");
    scanf("%s", username);
    
    printf("Enter the password: ");
    scanf("%s", password);
    
    // Case-insensitive check for "admin"
    if(username[0] != 'A' && username[0] != 'a' ||
       username[1] != 'D' && username[1] != 'd' ||
       username[2] != 'M' && username[2] != 'm' ||
       username[3] != 'I' && username[3] != 'i' ||
       username[4] != 'N' && username[4] != 'n' ||
       username[5] != '\0') {
        isValid = 0;
    }
    
    // Check password
    if(isValid && strcmp(password, "karunya") == 0) {
        printf("Login Successful \n");
    } else {
        printf("Invalid username or password\n");
    }
    
    return 0;
}
```

Each version offers a different method for handling the case-insensitive comparison of the username while ensuring the password matches exactly.
