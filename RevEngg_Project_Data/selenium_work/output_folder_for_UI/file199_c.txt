### Version 1:
```c
#include<stdio.h>

int main() {
    float heightInMeters, weight, bmi;

    printf("Enter your height in cm: ");
    scanf("%f", &heightInMeters);
    printf("Enter your weight in kg: ");
    scanf("%f", &weight);

    heightInMeters = heightInMeters / 100; // Convert cm to meters
    bmi = weight / (heightInMeters * heightInMeters); // Calculate BMI

    printf("Your BMI: %.2f\n", bmi);

    if(bmi < 18.5) {
        printf("Underweight\n");
    } else if(bmi < 25) {
        printf("Normal weight\n");
    } else if(bmi < 30) {
        printf("Overweight\n");
    } else {
        printf("Obesity\n");
    }

    return 0;
}
```

### Version 3:
```c
#include<stdio.h>

int main() {
    float heightM, weight, bmi;

    printf("Enter height in centimeters: ");
    scanf("%f", &heightM);
    printf("Enter weight in kilograms: ");
    scanf("%f", &weight);

    heightM = heightM / 100; // Height conversion
    bmi = weight / (heightM * heightM); // BMI formula

    printf("Calculated BMI: %.2f\nCategory: ", bmi);

    if(bmi <= 18.5) {
        printf("Underweight");
    } else if(bmi > 18.5 && bmi <= 24.9) {
        printf("Normal");
    } else if(bmi >= 25 && bmi <= 29.9) {
        printf("Overweight");
    } else {
        printf("Obese");
    }

    printf("\n");

    return 0;
}
```
