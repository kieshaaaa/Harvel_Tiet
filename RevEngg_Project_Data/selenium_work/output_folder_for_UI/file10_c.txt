### Version 2: Using While Loops

This version employs `while` loops instead of `for` loops for iteration, offering a different loop structure for achieving the same pattern.

```c
#include <stdio.h>

int main() {
    int n;
    printf("Enter the number of rows: ");
    scanf("%d", &n);

    if (n < 1) {
        printf("Please enter a positive number.");
        return 0;
    }

    int i = 1;
    while (i <= n) {
        int space = 1;
        while (space < i) {
            printf("  ");
            space++;
        }

        int num = i;
        while (num <= n) {
            printf("%d ", num);
            num++;
        }

        printf("\n");
        i++;
    }

    return 0;
}
```
