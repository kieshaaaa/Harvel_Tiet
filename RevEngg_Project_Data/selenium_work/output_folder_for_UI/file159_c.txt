### Version 1: Change in Variable Names and Values

```c
#include <stdio.h>

int x=15; // global variable

void display();

int main() {
    int x=25; // local to main
    int y=35; // local to main

    printf("In main()  x=%d, y=%d\n", x, y);
    display();
    printf("In main() after calling display() ~ y=%d\n", y);
    return 0;
}

void display() {
    int y=45; // local to display

    printf("In display()  x= %d\n", x);
    printf("In display()  y= %d\n", y);
}
```

### Version 3: Reusing `a` and `b` with Different Scopes and a Calculation

```c
#include <stdio.h>

int a=7; // global variable

void process();

int main() {
    int a=12; // local to main
    int b=22; // local to main
    int sum;

    sum = a + b;
    printf("In main()  a=%d, b=%d, sum=%d\n", a, b, sum);
    process();
    sum = b - a;
    printf("In main() after calling process() ~ sum=%d\n", sum);
    return 0;
}

void process() {
    int b=32; // local to process
    int diff;

    diff = b - a;
    printf("In process()  a= %d\n", a);
    printf("In process()  b= %d, diff=%d\n", b, diff);
}
```
