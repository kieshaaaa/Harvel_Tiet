### Version 1: Simplified Approach

```c
#include <stdio.h>

int main() {
    int rows;
    printf("Enter the number of rows: ");
    scanf("%d", &rows);
    if (rows < 1) {
        printf("Please enter a positive number.\n");
        return 0;
    }

    int cols = 2 * rows - 1;

    for (int i = 0; i < rows; i++) {
        for (int j = 0; j < cols; j++) {
            if (i + j == rows - 1 || j - i == rows - 1 || i == 0) {
                printf("*");
            } else {
                printf(" ");
            }
        }
        printf("\n");
    }

    return 0;
}
```

### Version 2: Iterative Reduction

```c
#include <stdio.h>

int main() {
    int n;
    printf("Enter the height of the pattern: ");
    scanf("%d", &n);
    if (n < 1) {
        printf("Invalid input. Please enter a positive integer.\n");
        return 0;
    }

    for (int row = 0; row < n; row++) {
        // Spaces
        for (int space = 0; space < row; space++) {
            printf(" ");
        }
        // Stars and internal spaces
        for (int col = 0; col < 2 * (n - row) - 1; col++) {
            if (col == 0 || col == 2 * (n - row) - 2 || row == 0) {
                printf("*");
            } else {
                printf(" ");
            }
        }
        printf("\n");
    }

    return 0;
}
```

### Version 3: Compact and Efficient

```c
#include <stdio.h>

int main() {
    int lines;
    printf("Enter number of lines: ");
    scanf("%d", &lines);
    if (lines < 1) {
        printf("Invalid input. Please enter a positive number.\n");
        return 0;
    }

    int maxWidth = 2 * lines - 1;

    for (int i = 1; i <= lines; i++) {
        // Initial spaces
        for (int j = 1; j < i; j++) {
            printf(" ");
        }
        // Print pattern
        for (int k = 1; k <= maxWidth - 2 * (i - 1); k++) {
            if (i == 1 || k == 1 || k == maxWidth - 2 * (i - 1)) {
                printf("*");
            } else {
                printf(" ");
            }
        }
        printf("\n");
    }

    return 0;
}
```

Each of these versions approaches the problem a bit differently but achieves the same goal of printing the specified pattern. Note that I followed your instructions to not use features you've excluded.
