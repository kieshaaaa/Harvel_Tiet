### Version 1: Swapping the Usage of Variables

```c
#include <stdio.h>

int x = 50; // Global variable

void display();

int main() {
    int x = 25; // Local to main
    int y = 35; // Local to main

    printf("In main() x=%d, y=%d\n", x, y);
    display();
    printf("Back in main() y=%d\n", y);
    return 0;
}

void display() {
    int y = 45; // Local to display

    printf("In display() x=%d\n", x);
    printf("In display() y=%d\n", y);
}
```

### Version 3: Using Functions to Modify Global Variables

```c
#include <stdio.h>

int counter = 5; // Global variable

    int localCounter = 10; // Local to main

    printf("In main() global counter is now=%d\n", counter);
    printf("In main() anotherCounter=%d\n", anotherCounter);

void incrementCounter() {
    // Here, we directly modify the global variable
    counter += 5;
    int tempCounter = 20; // Local to incrementCounter

}
```
