### Version 1: Using While Loop for Factorial Calculation

```c
#include <stdio.h>

int main() {
    int lower, upper, j, i;
    float result;

    printf("Enter the lower bound: ");
    scanf("%d", &lower);
    printf("Enter the upper bound: ");
    scanf("%d", &upper);

    for(j = lower; j <= upper && j >= 0; j++) {
        i = 2;
        result = 1.0;
        while(i <= j && result <= upper) {
            result *= i;
            if (result < 0) { // Checking for overflow
                result = -1;
                break;
            }
            i++;
        }
        
        if(result <= upper && result > 0) {
            printf("Factorial of %d is %.0f\n", j, result);
        } else {
            break;
        }
    }

    return 0;
}
```

### Version 2: Inline Factorial Calculation Inside the Loop

```c
#include <stdio.h>

int main() {
    int lower, upper;
    printf("Enter the lower bound: ");
    scanf("%d", &lower);
    printf("Enter the upper bound: ");
    scanf("%d", &upper);

    for (int j = lower; j <= upper && j >= 0; j++) {
        float fact = 1.0;
        for (int i = 2; i <= j; i++) {
            fact *= i;
            if (fact < 0) { // Detecting overflow
                fact = -1;
                break;
            }
        }
        
        if (fact <= upper && fact > 0) {
            printf("Factorial of %d is %.0f\n", j, fact);
        } else {
            break;
        }
    }

    return 0;
}
```

### Version 3: Separating Factorial Calculation with a Flag for Overflow

```c
#include <stdio.h>

int main() {
    int lower, upper;
    printf("Enter the lower bound: ");
    scanf("%d", &lower);
    printf("Enter the upper bound: ");
    scanf("%d", &upper);

    for (int j = lower; j <= upper && j >= 0; j++) {
        float fact = 1.0;
        int overflow = 0; // Flag to indicate overflow
        for (int i = 2; i <= j; i++) {
            fact *= i;
            if (fact < 0) { // Checking for overflow
                overflow = 1;
                break;
            }
        }
        
        if (!overflow && fact <= upper && fact > 0) {
            printf("Factorial of %d is %.0f\n", j, fact);
        } else {
            break;
        }
    }

    return 0;
}
```
