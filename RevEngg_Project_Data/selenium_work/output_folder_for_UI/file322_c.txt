### Version 1: Iterative Approach Using For Loop

```c
#include <stdio.h>

int main() {
    int num1, num2, min, hcf = 1;

    printf("Enter first number :: ");
    scanf("%d", &num1);
    printf("\nEnter second number :: ");
    scanf("%d", &num2);

    min = (num1 < num2) ? num1 : num2;

    for (int i = 2; i <= min; i++) {
        if (num1 % i == 0 && num2 % i == 0) {
            hcf = i;
        }
    }

    printf("\nHCF (Highest Common Factor) of %d,%d is: %d\n", num1, num2, hcf);

    return 0;
}
```

### Version 2: While Loop Without Temporary Variable

```c
#include <stdio.h>

int main() {
    int num1, num2;

    printf("Enter first number :: ");
    scanf("%d", &num1);
    printf("\nEnter second number :: ");
    scanf("%d", &num2);

    while (num1 != num2) {
        if (num1 > num2)
            num1 -= num2;
        else
            num2 -= num1;
    }

    printf("\nHCF (Highest Common Factor) of %d,%d is: %d\n", num1 + num2 - num1, num1 + num2 - num2, num1); // Avoiding expressions in printf

    return 0;
}
```

### Version 3: Simplified Iterative Approach

```c
#include <stdio.h>

int main() {
    int a, b, hcf;

    printf("Enter first number :: ");
    scanf("%d", &a);
    printf("\nEnter second number :: ");
    scanf("%d", &b);

    for (hcf = (a < b) ? a : b; hcf > 0; hcf--) {
        if (a % hcf == 0 && b % hcf == 0)
            break;
    }

    printf("\nHCF (Highest Common Factor) of %d,%d is: %d\n", a, b, hcf);

    return 0;
}
```

Each version offers a different way to calculate the HCF of two numbers, following your constraints.
