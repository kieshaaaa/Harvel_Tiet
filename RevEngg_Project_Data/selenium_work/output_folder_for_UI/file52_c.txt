### Version 1: Using For Loop

```c
#include <stdio.h>

void convertToLowercase(char s[]);
void convertToUppercase(char s[]);

int main() {
    char str[100];

    printf("Enter any string : ");
    scanf("%[^\n]s", str);

    convertToLowercase(str);
    printf("String in lowercase : %s\n", str);

    convertToUppercase(str);
    printf("String in uppercase : %s\n", str);

    return 0;
}

void convertToLowercase(char s[]) {
    for(int i = 0; s[i] != '\0'; i++) {
        if(s[i] >= 'A' && s[i] <= 'Z') {
            s[i] += 'a' - 'A';
        }
    }
}

void convertToUppercase(char s[]) {
    for(int i = 0; s[i] != '\0'; i++) {
        if(s[i] >= 'a' && s[i] <= 'z') {
            s[i] -= 'a' - 'A';
        }
    }
}
```

### Version 2: Using While Loop and Conditional Operator

```c
#include <stdio.h>

void toLowercase(char s[]);
void toUppercase(char s[]);

int main() {
    char str[100];

    printf("Enter any string : ");
    scanf("%[^\n]s", str);

    toLowercase(str);
    printf("Lowercase string : %s\n", str);

    toUppercase(str);
    printf("Uppercase string : %s\n", str);

    return 0;
}

void toLowercase(char s[]) {
    int i = 0;
    while (s[i]) {
        s[i] = (s[i] >= 'A' && s[i] <= 'Z') ? (s[i] + 32) : s[i];
        i++;
    }
}

void toUppercase(char s[]) {
    int i = 0;
    while (s[i]) {
        s[i] = (s[i] >= 'a' && s[i] <= 'z') ? (s[i] - 32) : s[i];
        i++;
    }
}
```

### Version 3: Compact For Loop Implementation

```c
#include <stdio.h>

void makeLowercase(char s[]);
void makeUppercase(char s[]);

int main() {
    char str[100];

    printf("Enter any string : ");
    scanf("%[^\n]s", str);

    makeLowercase(str);
    printf("String in lowercase : %s\n", str);

    makeUppercase(str);
    printf("String in uppercase : %s\n", str);

    return 0;
}

void makeLowercase(char s[]) {
    for (int i = 0; s[i]; i++) {
        s[i] += (s[i] >= 'A' && s[i] <= 'Z') ? 32 : 0;
    }
}

void makeUppercase(char s[]) {
    for (int i = 0; s[i]; i++) {
        s[i] -= (s[i] >= 'a' && s[i] <= 'z') ? 32 : 0;
    }
}
```
