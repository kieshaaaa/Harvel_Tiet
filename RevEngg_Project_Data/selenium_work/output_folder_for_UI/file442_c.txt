### Version 1
```c
#include <stdio.h>

int main() {
    float temperature, convertedTemperature;
    int choice;

    printf("\n1: Fahrenheit to Celsius.");
    printf("\n2: Celsius to Fahrenheit.\n");
    printf("\nChoice (1/2): ");
    scanf("%d", &choice);

    if (choice == 1) {
        printf("\nEnter Fahrenheit: ");
        scanf("%f", &temperature);
        convertedTemperature = (temperature - 32) / 1.8;
        printf("\nCelsius: %.2f", convertedTemperature);
    } else if (choice == 2) {
        printf("\nEnter Celsius: ");
        scanf("%f", &temperature);
        convertedTemperature = (temperature * 1.8) + 32;
        printf("\nFahrenheit: %.2f", convertedTemperature);
    } else {
        printf("\nInvalid choice.");
    }
    return 0;
}
```

```c
int main() {
```

```c
    printf("\nOption (F/C): ");
    scanf(" %c", &option); // Notice the space before %c to consume any newline left in the input buffer

    if (option == 'F' || option == 'f') {
        printf("\nFahrenheit: ");
        scanf("%f", &temperatureF);
        temperatureC = (temperatureF - 32) * 5.0 / 9.0;
        printf("\nCelsius: %.2f", temperatureC);
    } else if (option == 'C' || option == 'c') {
        printf("\nCelsius: ");
        scanf("%f", &temperatureC);
        temperatureF = (temperatureC * 9.0 / 5.0) + 32;
        printf("\nFahrenheit: %.2f", temperatureF);
    } else {
        printf("\nInvalid selection.");
    }
    return 0;
}
```
