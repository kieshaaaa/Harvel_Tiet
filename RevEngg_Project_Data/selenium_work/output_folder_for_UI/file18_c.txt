### Version 1: Using else-if Ladder

```c
#include <stdio.h>

int main() {
    int x, y, z, largest;

    printf("Enter three numbers (separated by space): ");
    scanf("%d %d %d", &x, &y, &z);

    largest = x; // Assume x is the largest first

    if(y > largest)       
        largest = y;
    if(z > largest)      
        largest = z;

    printf("Largest number is = %d\n", largest);

    return 0;
}
```

### Version 2: Using Nested if Statements

```c
#include <stdio.h>

int main() {
    int x, y, z, largest;

    printf("Enter three numbers (separated by space): ");
    scanf("%d %d %d", &x, &y, &z);

    if(x > y) {
        if(x > z) 
            largest = x;
        else 
            largest = z;
    } else {
        if(y > z) 
            largest = y;
        else 
            largest = z;
    }

    printf("Largest number is = %d\n", largest);

    return 0;
}
```

### Version 3: Using Conditional Operator

```c
#include <stdio.h>

int main() {
    int x, y, z, largest;

    printf("Enter three numbers (separated by space): ");
    scanf("%d %d %d", &x, &y, &z);

    largest = (x > y) ? (x > z ? x : z) : (y > z ? y : z);

    printf("Largest number is = %d\n", largest);

    return 0;
}
```
