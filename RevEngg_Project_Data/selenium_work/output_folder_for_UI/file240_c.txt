### Version 1: Inline Calculations Without Separate Functions

```c
#include <stdio.h>
#include <math.h>

int main() {
    int n, num = 1, count = 0;
    printf("Enter the value of n to find the nth Armstrong number: ");
    scanf("%d", &n);

    while (1) {
        int temp = num, sum = 0, digits = 0, t = temp;
        // Counting digits
        while (t != 0) {
            t /= 10;
            digits++;
        }
        // Calculating sum of powers of digits
        t = temp;
        while (t != 0) {
            int remainder = t % 10;
            sum += pow(remainder, digits);
            t /= 10;
        }
        // Checking for Armstrong number
        if (num == sum) {
            count++;
            if (count == n) {
                printf("The %dth Armstrong number is: %d\n", n, num);
                break;
            }
        }
        num++;
    }
    return 0;
}
```

### Version 2: Reduced Scope Variables

```c
#include <stdio.h>
#include <math.h>

int main() {
    int n;
    printf("Enter the value of n to find the nth Armstrong number: ");
    scanf("%d", &n);

    for (int num = 1, count = 0;; num++) {
        int sum = 0, temp = num;
        float digits = 0;
        for (int t = temp; t; digits++, t /= 10);
        
        for (int t = temp; t; t /= 10) {
            sum += pow(t % 10, (int)digits);
        }
        
        if (sum == num) {
            if (++count == n) {
                printf("The %dth Armstrong number is: %d\n", n, num);
                break;
            }
        }
    }

    return 0;
}
```

### Version 3: Simplifying and Integrating Loops

```c
#include <stdio.h>
#include <math.h>

int main() {
    int n, num = 1;
    printf("Enter the value of n to find the nth Armstrong number: ");
    scanf("%d", &n);

    for (int count = 0; count < n; num++) {
        int sum = 0, temp = num, digits = 0;
        // Inline digit counting and sum calculation
        for (int t = temp; t; t /= 10, digits++);
        for (int t = temp; t; t /= 10) {
            sum += pow(t % 10, digits);
        }
        // Checking and counting Armstrong number
        if (sum == num) {
            count++;
            if (count == n) {
                printf("The %dth Armstrong number is: %d\n", n, num);
                break;
            }
        }
    }

    return 0;
}
```
