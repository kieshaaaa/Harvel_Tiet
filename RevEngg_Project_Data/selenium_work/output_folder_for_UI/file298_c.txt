Given your specific requirements, I'll create three different versions of the program to count the number of characters in a string. Each version will have slight variations in approach while adhering to your constraints.

### Version 1: Using `scanf` instead of `gets`

```c
#include<stdio.h>

int main() {
    char str[20];
    int i = 0, count = 0;
    printf("\nEnter the string: ");
    scanf("%19[^\n]", str); // Safe alternative to gets, limits input size, excludes newline

    while (str[i] != '\0') {
        count++;
        i++;
    }

    printf("\nLength of [ ");
    printf("%s", str);
    printf(" ] is [ ");
    printf("%d", count);
    printf(" ]\n");
    return 0;
}
```

### Version 2: Incorporating a `for` loop

```c
#include<stdio.h>

int main() {
    char str[20];
    int count = 0;
    printf("\nEnter the string: ");
    scanf("%19[^\n]", str); // Using scanf with a limit for input size

    for (int i = 0; str[i] != '\0'; i++) {
        count++;
    }

    printf("\nLength of [ ");
    printf("%s", str);
    printf(" ] is [ ");
    printf("%d", count);
    printf(" ]\n");
    return 0;
}
```

### Version 3: Using `fgets` for safer input

```c
#include<stdio.h>

int main() {
    char str[20];
    int i = 0;
    printf("\nEnter the string: ");
    fgets(str, sizeof(str), stdin); // Safe way to read strings

    // Remove newline character if present
    while(str[i] != '\0') {
        if (str[i] == '\n') {
            str[i] = '\0';
            break;
        }
        i++;
    }

    // Reset i for count calculation
    i = 0;
    int count = 0;
    while (str[i] != '\0') {
        count++;
        i++;
    }

    printf("\nLength of [ ");
    printf("%s", str);
    printf(" ] is [ ");
    printf("%d", count);
    printf(" ]\n");
    return 0;
}
```

These versions address your constraints while providing alternative ways to count the characters in a string. Each uses different functions for string input (`scanf`, `fgets`) and different loop structures (`while`, `for`).
