### Version 1: Bitwise OR Operation

```c
#include <stdio.h>

int main() {
    int a, b, result;
    printf("Enter the first integer (a): ");
    scanf("%d", &a);
    printf("Enter the second integer (b): ");
    scanf("%d", &b);

    result = a | b;

    printf("The result is %d \n", result);

    return 0;
}
```

### Version 2: Bitwise XOR Operation

```c
#include <stdio.h>

int main() {
    int a, b, result;
    printf("Enter the first integer (a): ");
    scanf("%d", &a);
    printf("Enter the second integer (b): ");
    scanf("%d", &b);

    result = a ^ b;

    printf("The result is %d \n", result);

    return 0;
}
```

### Version 3: Bitwise Shift Left Operation

```c
#include <stdio.h>

int main() {
    int a, shift;
    printf("Enter an integer (a): ");
    scanf("%d", &a);
    printf("Enter number of positions to shift left: ");
    scanf("%d", &shift);

    int result = a << shift;

    printf("The result is %d \n", result);

    return 0;
}
```
