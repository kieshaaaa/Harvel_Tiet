### Version 1: Subtraction Method for HCF

```c
#include <stdio.h>
#include <math.h>

int main() {
    int num1, num2;
    printf("\nEnter Two Numbers to Find HCF => ");
    scanf("%d%d", &num1, &num2);

    num1 = fabs(num1);
    num2 = fabs(num2);

    while(num1 != num2) {
        if(num1 > num2)
            num1 -= num2;
        else
            num2 -= num1;
    }

    printf("\nHCF => %d\n", num1);
    return 0;
}
```

### Version 2: Iterative Euclidean Algorithm

```c
#include <stdio.h>
#include <math.h>

int main() {
    int num1, num2, temp1, temp2, remainder;
    printf("\nEnter Two Numbers to Find HCF => ");
    scanf("%d%d", &num1, &num2);

    num1 = fabs(num1);
    num2 = fabs(num2);
    temp1 = num1;
    temp2 = num2;

    while(temp2 != 0) {
        remainder = temp1 % temp2;
        temp1 = temp2;
        temp2 = remainder;
    }

    printf("\nHCF => %d\n", temp1);
    return 0;
}
```

### Version 3: Brute Force Method

```c
#include <stdio.h>
#include <math.h>

int main() {
    int num1, num2, hcf = 1;
    printf("\nEnter Two Numbers to Find HCF => ");
    scanf("%d%d", &num1, &num2);

    num1 = fabs(num1);
    num2 = fabs(num2);
    int min = num1 < num2 ? num1 : num2;

    for(int i = 1; i <= min; i++) {
        if(num1 % i == 0 && num2 % i == 0) {
            hcf = i;
        }
    }

    printf("\nHCF => %d\n", hcf);
    return 0;
}
```

Each of these versions employs a different method to calculate the HCF of two numbers. The first uses the subtraction method, the second employs the iterative Euclidean algorithm, and the third is a brute force method by checking each number from 1 to the minimum of the two numbers.
