int main() {

    if (gender == 'M' || gender == 'm') {
        if (age < 18) {
            printf("Boy\n");
        } else {
            printf("Man\n");
        }

        if (age < 18) {
            printf("Girl\n");
        } else {
            printf("Woman\n");
        }

    }

}
