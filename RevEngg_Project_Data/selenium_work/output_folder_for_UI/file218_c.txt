### Version 1: Using a While Loop

#include <stdio.h>

int main() {
    int a, b;
    int mul = 0;
    int count = 0;

    printf("Enter first number: ");
    scanf("%d", &a);
    printf("Enter second number: ");
    scanf("%d", &b);

    while (count < b) {
        mul += a;
        count++;
