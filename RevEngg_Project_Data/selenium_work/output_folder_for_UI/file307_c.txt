### Version 1: Simplified `if-else` Structure
```c
#include <stdio.h>
#include <stdlib.h>
#include <math.h>

int main() {
    float a, b, c, discriminant, root1, root2, imgPart;

    printf("\nEnter Coefficients a, b and c: ");
    scanf("%f%f%f", &a, &b, &c);

    if (a == 0) {
        printf("\nNot a quadratic equation.");
        return 0;
    }

    discriminant = b * b - 4 * a * c;
    printf("\nDiscriminant: %f", discriminant);

    if (discriminant > 0) {
        root1 = (-b + sqrt(discriminant)) / (2 * a);
        root2 = (-b - sqrt(discriminant)) / (2 * a);
        printf("\nRoots are real and distinct: %f and %f", root1, root2);
    } else if (discriminant == 0) {
        root1 = -b / (2 * a);
        printf("\nRoots are real and equal: %f and %f", root1, root1);
    } else {
        root1 = -b / (2 * a);
        imgPart = sqrt(-discriminant) / (2 * a);
        printf("\nRoots are complex: %f + i%f and %f - i%f", root1, imgPart, root1, imgPart);
    }

    return 0;
}
```

### Version 2: Using Ternary Operator for Discriminant Check
```c
#include <stdio.h>
#include <stdlib.h>
#include <math.h>

int main() {
    float a, b, c, discriminant, root1, root2, imgPart;

    printf("\nEnter Coefficients a, b and c: ");
    scanf("%f%f%f", &a, &b, &c);

    if (a == 0) {
        printf("\nNot a quadratic equation.");
        return 0;
    }

    discriminant = b * b - 4 * a * c;
    printf("\nDiscriminant: %f", discriminant);

    discriminant > 0 ? (
        root1 = (-b + sqrt(discriminant)) / (2*a),
        root2 = (-b - sqrt(discriminant)) / (2*a),
        printf("\nRoots are real and distinct: %f and %f", root1, root2)
    ) : (
        discriminant == 0 ? (
            root1 = -b / (2*a),
            printf("\nRoots are real and equal: %f and %f", root1, root1)
        ) : (
            root1 = -b / (2*a),
            imgPart = sqrt(-discriminant) / (2*a),
            printf("\nRoots are complex: %f + i%f and %f - i%f", root1, imgPart, root1, imgPart)
        )
    );

    return 0;
}
```

### Version 3: Minimizing Variables
```c
#include <stdio.h>
#include <math.h>

int main() {
    float a, b, c, discriminant;

    printf("\nEnter Coefficients a, b and c: ");
    scanf("%f%f%f", &a, &b, &c);

    if (a == 0) {
        printf("\nNot a quadratic equation.");
        return 0;
    }

    discriminant = b*b - 4*a*c;
    printf("\nDiscriminant: %f", discriminant);

    if (discriminant > 0) {
        printf("\nRoots are real and distinct: %f and %f",
               (-b + sqrt(discriminant)) / (2*a),
               (-b - sqrt(discriminant)) / (2*a));
    } else if (discriminant == 0) {
        printf("\nRoots are real and equal: %f and %f",
               -b / (2*a), -b / (2*a));
    } else {
        printf("\nRoots are complex: %f + i%f and %f - i%f",
               -b / (2*a), sqrt(-discriminant) / (2*a),
               -b / (2*a), sqrt(-discriminant) / (2*a));
    }

    return 0;
}
```
