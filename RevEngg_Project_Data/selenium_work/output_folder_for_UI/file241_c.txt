### Version 1: Simplified Input and More Direct Output

```c
#include <stdio.h>

int main() {
    int n, search;
    int array[100]; // Assuming the size is sufficient for the task
    
    // Simplified input process
    printf("Number of elements: ");
    scanf("%d", &n);
    
    printf("Enter elements: ");
    for (int i = 0; i < n; i++) {
        scanf("%d", &array[i]);
    }
    
    printf("Search value: ");
    scanf("%d", &search);
    
    int first = 0, last = n - 1;
    int found = 0; // To mark if the element is found
    
    while (first <= last) {
        int middle = (first + last) / 2;
        
        if (array[middle] == search) {
            printf("%d found at position %d.\n", search, middle + 1);
            found = 1;
            break;
        } else if (array[middle] < search) {
            first = middle + 1;
        } else {
            last = middle - 1;
        }
    }
    
    if (!found) {
        printf("Not found! %d is not present in the list.\n", search);
    }
    
    return 0;
}
```

### Version 2: Loop Optimizations and Variable Reduction

```c
#include <stdio.h>

int main() {
    int n, search;
    int array[100];
    
    printf("Elements count: ");
    scanf("%d", &n);
    
    for (int i = 0; i < n; i++) {
        printf("Element %d: ", i + 1);
        scanf("%d", &array[i]);
    }
    
    printf("Search for: ");
    scanf("%d", &search);
    
    for (int first = 0, last = n - 1, middle; first <= last;) {
        middle = first + (last - first) / 2;
        
        if (array[middle] == search) {
            printf("%d is at position %d.\n", search, middle + 1);
            return 0;
        }
        
        if (array[middle] < search) first = middle + 1;
        else last = middle - 1;
    }
    
    printf("%d not found in the list.\n", search);
    
    return 0;
}
```

### Version 3: Iterative Approach with Less Interactivity

```c
#include <stdio.h>

int main() {
    int array[100], n, search;
    
    printf("Enter total elements: ");
    scanf("%d", &n);
    
    printf("Input elements in ascending order: ");
    for (int i = 0; i < n; ++i) {
        scanf("%d", &array[i]);
    }
    
    printf("Element to find: ");
    scanf("%d", &search);
    
    int low = 0, high = n - 1, middle;
    
    while (low <= high) {
        middle = low + (high - low) / 2;
        
        if (array[middle] == search) {
            printf("Element %d found at index %d.\n", search, middle);
            return 0;
        } else if (array[middle] < search) {
            low = middle + 1;
        } else {
            high = middle - 1;
        }
    }
    
    printf("Element %d not found.\n", search);
    
    return 0;
}
```
