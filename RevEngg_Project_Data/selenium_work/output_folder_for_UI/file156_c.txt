### Version 1: Simplified Loop Structure

```c
#include <stdio.h>

int main() {
    int arr[10], i;
    
    printf("Enter 10 elements of the array:\n");
    for (i = 0; i < 10; i++) {
        printf("Element %d: ", i);
        scanf("%d", &arr[i]);
    }
    
    printf("\nOriginal Array:\n");
    for (i = 0; i < 10; i++) {
        printf("Element %d: %d\n", i, arr[i]);
    }
    
    for (i = 0; i < 10; i++) {
        arr[i] = arr[i] % 2;
    }
    
    printf("\nModified Array (Even=0, Odd=1):\n");
    for (i = 0; i < 10; i++) {
        printf("Element %d: %d\n", i, arr[i]);
    }
    
    return 0;
}
```

### Version 2: Array Initialization at Declaration

```c
#include <stdio.h>

int main() {
    int arr[10], i;
    printf("Populate array with 10 elements:\n");
    
    for (i = 0; i < 10; ++i) {
        printf("Enter arr[%d]: ", i);
        scanf("%d", &arr[i]);
    }
    
    printf("\nBefore Replacing:\n");
    for (i = 0; i < 10; ++i) {
        printf("arr[%d]: %d\n", i, arr[i]);
    }
    
    for (i = 0; i < 10; ++i) {
        arr[i] = (arr[i] % 2) ? 1 : 0;
    }
    
    printf("\nAfter Replacing (0=Even, 1=Odd):\n");
    for (i = 0; i < 10; ++i) {
        printf("arr[%d]: %d\n", i, arr[i]);
    }
    
    return 0;
}
```

### Version 3: Using a Single Loop for Initialization, Replacement, and Display

```c
#include <stdio.h>

int main() {
    int arr[10], i;
    printf("Enter 10 integers:\n");
    
    for (i = 0; i < 10; i++) {
        printf("Number %d: ", i);
        scanf("%d", &arr[i]);
        arr[i] = arr[i] % 2; // Immediate replacement upon input
    }
    
    printf("\nProcessed Array (Even as 0, Odd as 1):\n");
    for (i = 0; i < 10; i++) {
        printf("Index %d: %d\n", i, arr[i]);
    }
    
    return 0;
}
```

/* C Program to replace all Even elements by 0 and Odd by 1  */

#include <stdio.h>

void readArray(int arr[], int size)
{
    int i =0;

    printf("Enter elements of array :: \n");

    for(i=0; i < size; i++)
    {
        printf("\nEnter arr[%d] :: ",i);
        scanf("%d",&arr[i]);
    }
}

void printArray(int arr[], int size)
{
    int i =0;

    printf("\nElements are : ");

    for(i=0; i < size; i++)
    {
        printf("\narr[%d] : %d",i,arr[i]);
    }
    printf("\n");
}

void replaceEvenOdd(int arr[], int size)
{
    int i=0;

    for(i=0; i < size; i++)
    {
        if( arr[i] % 2 == 0 )
            arr[i] = 0 ;
        else
            arr[i] = 1 ;
    }
}

int main()
{
    int arr[10];

    readArray(arr,10);

    printf("\nBefore replacement : ");
    printArray(arr,10);

    replaceEvenOdd(arr,10);

    printf("\nAfter replacement : ");
    printArray(arr,10);

    return 0;
}
