### Version 1: Using While Loops

```c
#include<stdio.h>

int main() {
   int i = 0, j;
   int count = 1;

   while (i <= 8) {
      printf("\n");
      j = 0;
      while (j < i) {
         printf("%d  ", count);
         count++;
         j++;
      }
      i++;
   }
   return 0;
}
```

### Version 2: Reversed Pyramid

```c
#include<stdio.h>

int main() {
   int i, j, space;
   int count = 1, rows = 8;

   for (i = rows; i >= 1; --i) {
      for (space = 0; space < rows - i; ++space)
         printf("   ");
      for (j = i; j <= 2 * i - 1; ++j) {
         printf("%d  ", count);
         count++;
      }
      for (j = 0; j < i - 1; ++j) {
         printf("%d  ", count);
         count++;
      }
      printf("\n");
   }
   return 0;
}
```

### Version 3: Fibonacci Series in Pyramid Shape

```c
#include<stdio.h>

int main() {
   int rows = 8;
   int a = 0, b = 1, c, i, j;

   for (i = 1; i <= rows; i++) {
      printf("\n");
      for (j = 1; j <= i; j++) {
         if (i == 1) {
            printf("%d  ", a);
            continue;
         }
         if (i == 2 && j == 1) {
            printf("%d  ", b);
            continue;
         }
         c = a + b;
         a = b;
         b = c;
         printf("%d  ", c);
      }
   }
   return 0;
}
```
