### Version 1: Using the Arithmetic Series Formula

```c
#include <stdio.h>

int main() {
    int n;
    float sum;  // Using float as per the instructions

    printf("Enter the value of N: ");
    scanf("%d", &n);

    // Arithmetic series formula to calculate the sum
    sum = (n / 2.0) * (1 + n); 

    printf("Sum is: %.0f\n", sum);  // Ensuring the output is treated as an integer

    return 0;
}
```

### Version 2: Using a While Loop

```c
#include <stdio.h>

int main() {
    int n, i = 1;
    float sum = 0;  // Using float as per the instructions

    printf("Enter the value of N: ");
    scanf("%d", &n);

    // Using a while loop instead of for loop
    while (i <= n) {
        sum += i;
        i++;
    }

    printf("Sum is: %.0f\n", sum);  // Ensuring the output is treated as an integer

    return 0;
}
```

### Version 3: Using the Do-While Loop

```c
#include <stdio.h>

int main() {
    int n, i = 1;
    float sum = 0;  // Using float as per the instructions

    printf("Enter the value of N: ");
    scanf("%d", &n);

    // Using a do-while loop
    do {
        sum += i;
        i++;
    } while (i <= n);

    printf("Sum is: %.0f\n", sum);  // Ensuring the output is treated as an integer

    return 0;
}
```

The second and third versions demonstrate alternate looping constructs to achieve the same outcome.
