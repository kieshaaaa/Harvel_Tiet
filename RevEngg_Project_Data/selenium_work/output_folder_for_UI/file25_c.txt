### Version 1: Enhanced Input Validation

This version adds basic input validation to ensure that division by zero is handled gracefully.

```c
#include <stdio.h>

int main() {
    int num1, num2;
    float result;
    char ch;
    
    printf("Enter first number: ");
    scanf("%d", &num1);
    printf("Enter second number: ");
    scanf("%d", &num2);
    
    printf("Choose operation to perform (+,-,*,/,%%): ");
    scanf(" %c", &ch);
    
    switch(ch) {
        case '+':
            result = num1 + num2;
            printf("Result: %d\n", (int)result);
            break;
        case '-':
            result = num1 - num2;
            printf("Result: %d\n", (int)result);
            break;
        case '*':
            result = num1 * num2;
            printf("Result: %d\n", (int)result);
            break;
        case '/':
            if(num2 != 0) {
                result = (float)num1 / (float)num2;
                printf("Result: %.2f\n", result);
            } else {
                printf("Division by zero is not allowed.\n");
            }
            break;
        case '%':
            if(num2 != 0) {
                result = num1 % num2;
                printf("Result: %d\n", (int)result);
            } else {
                printf("Division by zero is not allowed.\n");
            }
            break;
        default:
            printf("Invalid operation.\n");
    }
    return 0;
}
```

### Version 2: Loop Until Valid Operation is Chosen

This version loops until the user inputs a valid operation, enhancing user experience by not requiring re-entry of numbers for a simple typo in the operation.

```c
#include <stdio.h>

int main() {
    int num1, num2;
    float result;
    char ch;
    int validOperation = 0;
    
    printf("Enter first number: ");
    scanf("%d", &num1);
    printf("Enter second number: ");
    scanf("%d", &num2);
    
    while(!validOperation) {
        printf("Choose operation to perform (+,-,*,/,%%): ");
        scanf(" %c", &ch);
        
        switch(ch) {
            case '+':
            case '-':
            case '*':
            case '/':
            case '%':
                validOperation = 1;
                break;
            default:
                printf("Invalid operation. Please try again.\n");
        }
    }
    
    // Perform the calculation after a valid operation is chosen
    switch(ch) {
        case '+':
            result = num1 + num2;
            break;
        case '-':
            result = num1 - num2;
            break;
        case '*':
            result = num1 * num2;
            break;
        case '/':
            result = num2 != 0 ? (float)num1 / num2 : 0; // Avoid division by zero
            if(num2 == 0) printf("Division by zero error.\n");
            break;
        case '%':
            result = num2 != 0 ? num1 % num2 : 0; // Avoid division by zero
            if(num2 == 0) printf("Division by zero error.\n");
            break;
    }
    
    if(num2 != 0 || ch != '/' && ch != '%') {
        printf("Result: %f\n", result);
    }
    return 0;
}
```

### Version 3: Simplified Output Handling

This version simplifies the output handling by using a single `printf` statement after the switch-case block for operations that result in an integer output and handling float separately.

```c
#include <stdio.h>

int main() {
    int num1, num2;
    float result;
    char ch;
    
    printf("Enter first number: ");
    scanf("%d", &num1);
    printf("Enter second number: ");
    scanf("%d", &num2);
    
    printf("Choose operation to perform (+,-,*,/,%%): ");
    scanf(" %c", &ch);
    
    int isFloat = 0; // Flag to check if result is float
    
    switch(ch) {
        case '+':
            result = num1 + num2;
            break;
        case '-':
            result = num1 - num2;
            break;
        case '*':
            result = num1 * num2;
            break;
        case '/':
            if(num2 != 0) {
                result = (float)num1 / num2;
                isFloat = 1;
            } else {
                printf("Division by zero is not allowed.\n");
                return 0;
            }
            break;
        case '%':
            result = num1 % num2;
            break;
        default:
            printf("Invalid operation.\n");
            return 0;
    }
    
    if(isFloat) {
        printf("Result: %.2f\n", result);
    } else {
        printf("Result: %d\n", (int)result);
    }
    
    return 0;
}
```

Each version provides a slightly different approach to improving usability, error handling, and output formatting.
