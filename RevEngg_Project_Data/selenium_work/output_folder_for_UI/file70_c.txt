### Version 1: Using a single loop for both input and copy operations

```c
#include <stdio.h>

int main() {
    int first[100], second[100];
    int i, size;

    printf("Enter the size of the array : ");
    scanf("%d", &size);

    for(i = 0; i < size; i++) {
        printf("Enter element %d of the array : ", i + 1);
        scanf("%d", &first[i]);
        // Directly copy during input
        second[i] = first[i];
    }

    printf("\nElements of first array are : ");
    for(i = 0; i < size; i++) {
        printf(" %d ", first[i]);
    }

    printf("\n\nElements of second array are : ");
    for(i = 0; i < size; i++) {
        printf(" %d ", second[i]);
    }

    return 0;
}
```

### Version 2: Using while loop for copying instead of for loop

```c
#include <stdio.h>

int main() {
    int first[100], second[100];
    int i, size;

    printf("Enter the size of the array : ");
    scanf("%d", &size);

    for(i = 0; i < size; i++) {
        printf("Enter element %d of the array : ", i + 1);
        scanf("%d", &first[i]);
    }

    i = 0;
    while(i < size) {
        second[i] = first[i];
        i++;
    }

    printf("\nElements of first array are : ");
    for(i = 0; i < size; i++) {
        printf(" %d ", first[i]);
    }

    printf("\n\nElements of second array are : ");
    for(i = 0; i < size; i++) {
        printf(" %d ", second[i]);
    }

    return 0;
}
```

### Version 3: Minimizing the code by integrating the display of elements within the copying loop

```c
#include <stdio.h>

int main() {
    int first[100], second[100];
    int i, size;

    printf("Enter the size of the array : ");
    scanf("%d", &size);

    for(i = 0; i < size; i++) {
        printf("Enter element %d of the array : ", i + 1);
        scanf("%d", &first[i]);
    }

    printf("\nElements of first array are : ");
    for(i = 0; i < size; i++) {
        second[i] = first[i]; // Copying element
        printf(" %d ", first[i]); // Display first array element
        if(i == size - 1) { // Switching to second array output
            printf("\n\nElements of second array are : ");
        }
    }

    for(i = 0; i < size; i++) {
        printf(" %d ", second[i]); // Display second array element
    }

    return 0;
}
```

These versions maintain the constraints specified, offering alternative methods to accomplish the task of copying one array to another and displaying the results, using different loop structures and integration strategies.
