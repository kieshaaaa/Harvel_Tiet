### Version 1: Direct Check for Composite Numbers

```c
#include <stdio.h>

int main() {
    int n, i, j;
    int composite;
    int count = 0;

    printf("Enter a number: ");
    scanf("%d", &n);

    for (i = 4; i <= n; i++) {
        composite = 0;
        for (j = 2; j < i; j++) {
            if (i % j == 0) {
                composite = 1;
                break;
            }
        }
        if (composite) {
            count++;
        }
    }

    printf("Number of composite numbers till %d: %d\n", n, count);
    return 0;
}
```

### Version 2: Counting by Reduction

```c
#include <stdio.h>

int main() {
    int n, i, j;
    int total = 0; // To count total numbers that are not prime (including 1)

    printf("Enter a number: ");
    scanf("%d", &n);

    // Initially assume all numbers are composite (excluding 1 and 2)
    total = n - 2; 

    for (i = 2; i <= n; i++) {
        for (j = 2; j*j <= i; j++) {
            if (i % j == 0) {
                break;
            }
            if (j + 1 > i / j) {
                total--; // This was actually prime, so decrement total
                break;
            }
        }
    }

    // Adjust for 1, which is neither prime nor composite by traditional definitions
    printf("Number of composite numbers till %d: %d\n", n, total + 1);
    return 0;
}
```

### Version 3: Skipping Even Numbers After 2

```c
#include <stdio.h>

int main() {
    int n, i, j;
    int count = 0;
    int isComposite;

    printf("Enter a number: ");
    scanf("%d", &n);

    // Directly counting 2 as prime, all even numbers as composite
    if (n > 2) count = (n - 2) / 2;

    for (i = 3; i <= n; i += 2) {
        isComposite = 0;
        for (j = 3; j*j <= i; j += 2) {
            if (i % j == 0) {
                isComposite = 1;
                break;
            }
        }
        if (isComposite) {
            count++; // This number is composite
        } else if (i == 2) {
            // This check is redundant given the loop starts at 3, kept for clarity
            // Prime numbers are not being counted directly in this approach
        }
    }

    // Adjusting the count for 1 and 2 based on input
    count += (n >= 2) ? 1 : 0;

    printf("Number of composite numbers till %d: %d\n", n, count);
    return 0;
}
```
