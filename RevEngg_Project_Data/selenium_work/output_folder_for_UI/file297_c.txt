### Version 1: Basic Iteration with Adjustment in Loop Conditions

```c
#include <stdio.h>

int main() {
    int maxRows;
    printf("\nHow Many Rows => ");
    scanf("%d", &maxRows);

    if (maxRows < 1) {
        printf("\n!!! Invalid Input, Please Enter Positive Number....\n");
        return 0;
    }

    printf("\n--------------------------------------------\n");

    for (int row = 1; row <= maxRows; row++) {
        for (int space = maxRows - row; space > 0; space--) {
            printf(" ");
        }

        for (int star = 1; star <= (2 * row - 1); star++) {
            printf("*");
        }

        printf("\n");
    }

    return 0;
}
```

### Version 2: Using Single Loop with Conditional Operators for Space and Star Printing

```c
#include <stdio.h>

int main() {
    int maxRows;
    printf("\nHow Many Rows => ");
    scanf("%d", &maxRows);

    if (maxRows < 1) {
        printf("\n!!! Invalid Input, Please Enter Positive Number....\n");
        return 0;
    }

    printf("\n--------------------------------------------\n");

    for (int row = 0; row < maxRows; row++) {
        int printPos = maxRows - row;
        for (int col = 1; col <= maxRows * 2 - 1; col++) {
            if (col >= printPos && col <= (maxRows * 2 - 1) - (printPos - 1)) {
                printf("*");
            } else {
                printf(" ");
            }
        }
        printf("\n");
    }

    return 0;
}
```

### Version 3: Leveraging Arithmetic Progression for Star Count Calculation

```c
#include <stdio.h>

int main() {
    int maxRows;
    printf("\nHow Many Rows => ");
    scanf("%d", &maxRows);

    if (maxRows < 1) {
        printf("\n!!! Invalid Input, Please Enter Positive Number....\n");
        return 0;
    }

    printf("\n--------------------------------------------\n");

    for (int row = 1; row <= maxRows; row++) {
        // Calculate number of spaces and stars for the current row
        int numSpaces = maxRows - row;
        int numStars = row * 2 - 1;

        // Print spaces
        for (int s = 0; s < numSpaces; s++) {
            printf(" ");
        }

        // Print stars
        for (int s = 0; s < numStars; s++) {
            printf("*");
        }

        printf("\n");
    }

    return 0;
}
```
