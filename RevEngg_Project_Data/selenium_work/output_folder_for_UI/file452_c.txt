### Version 1: Simplified Version

This version simplifies the loop conditions a bit and uses only `printf` for printing both stars and spaces.

```c
#include <stdio.h>

int main() {
    int rows;
    printf("\nEnter the number of rows: ");
    scanf("%d", &rows);

    if (rows < 1) {
        printf("\nInvalid input. Please enter a positive number.");
        return 0;
    }

    printf("\n");

    for (int i = 0; i < rows; i++) {
        for (int j = 0; j < rows; j++) {
            if (i == 0 || i == rows - 1 || j == 0 || j == rows - 1)
                printf("*");
            else
                printf(" ");
        }
        printf("\n");
    }

    return 0;
}
```

### Version 2: Using Ternary Operator

This version uses the ternary operator to decide whether to print a star or a space in a more compact way.

```c
#include <stdio.h>

int main() {
    int size;
    printf("\nEnter the size: ");
    scanf("%d", &size);

    if (size < 1) {
        printf("\nInvalid input. Please enter a positive number.");
        return 0;
    }

    printf("\n");

    for (int r = 0; r < size; r++) {
        for (int c = 0; c < size; c++) {
            printf("%c", (r == 0 || r == size - 1 || c == 0 || c == size - 1) ? '*' : ' ');
        }
        printf("\n");
    }

    return 0;
}
```
