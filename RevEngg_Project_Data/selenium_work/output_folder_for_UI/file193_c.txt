Given your constraints, the task is to manipulate and update array values without using pointers, dynamic memory allocation, structures, unions, recursion, file handling, modifying existing data types, and avoiding doubles in favor of floats where necessary. Here are three alternative versions of the program achieving similar functionality:

### Version 1: Using Array Indexes Directly

```c
#include <stdio.h>

int main() {
    int values[3];
    values[0] = 100; // Assigning values directly
    values[1] = 200;
    values[2] = 300;

    // Printing initial values
    printf("\tValue of \ta: %d \tb: %d \tc: %d\n", values[0], values[1], values[2]);

    // Adding 10 to each value
    values[0] += 10;
    values[1] += 10;
    values[2] += 10;

    // Printing updated values
    printf("\n\tAfter adding 10\n\n");
    printf("\tValue of \ta: %d \tb: %d \tc: %d\n", values[0], values[1], values[2]);

    return 0;
}
```

### Version 2: Using a For Loop for Operations

```c
#include <stdio.h>

int main() {
    int nums[3] = {100, 200, 300}; // Initializing array with values

    // Printing initial values
    for (int i = 0; i < 3; i++) {
        printf("\tValue of nums[%d]: %d\n", i, nums[i]);
    }

    // Adding 10 to each value
    for (int i = 0; i < 3; i++) {
        nums[i] += 10;
    }

    printf("\n\tAfter adding 10\n\n");
    // Printing updated values
    for (int i = 0; i < 3; i++) {
        printf("\tValue of nums[%d]: %d\n", i, nums[i]);
    }

    return 0;
}
```

### Version 3: Using a Single Variable for Input and Modification

```c
#include <stdio.h>

int main() {

    // Assigning and printing each variable's value separately

    printf("\tValue of \tc: %d\n", x);

    // Resetting x for addition operation
    x = 100 + 10; // Adding 10 to the initial value
    printf("\n\tAfter adding 10\n\n");
    printf("\tValue of \ta: %d\n", x);
    x = 200 + 10; // Repeating addition for the second value
    printf("\tValue of \tb: %d\n", x);
    x = 300 + 10; // Repeating addition for the third value
    printf("\tValue of \tc: %d\n", x);

    return 0;
}
```
